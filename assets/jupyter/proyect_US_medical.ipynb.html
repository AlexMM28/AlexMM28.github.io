models = ["Linear Regression", "Decision Tree", "KNN"]

    mean_absolute_error(y_test, y_pred_lr),
    mean_absolute_error(y_test, y_pred_tree),
    mean_absolute_error(y_test, y_pred_knn)
]

    r2_score(y_test, y_pred_lr),
